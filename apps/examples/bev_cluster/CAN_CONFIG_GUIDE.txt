/* Step 5: Implement decode in can_db.c */
void can_db_decode_motor(const uint8_t *data,
                         struct vehicle_data_s *vdata)
{
  vdata->motor_rpm = (data[SIGNAL_RPM_BYTE] << 8) |
                     data[SIGNAL_RPM_BYTE + 1];
  
  vdata->motor_torque = (int16_t)((data[SIGNAL_TORQUE_BYTE] << 8) |
                                   data[SIGNAL_TORQUE_BYTE + 1]);
}

/* Battery message (0x18FEF200) contains multiple signals:
 *
 * Byte 0-1: Voltage (16-bit, 0.1V)
 * Byte 2-3: Current (16-bit signed, 0.1A)
 * Byte 4:   Direction (0=Fwd, 1=Rev, 2=Neutral)
 * Byte 5:   State of Charge (0-100%)
 * Byte 6-7: Reserved
 *
 * All handled by single decoder: can_db_decode_battery()
 */

/****************************************************************************
 * SIGNAL EXTRACTION MACROS (optional helpers)
 ****************************************************************************/

/* You can add these to can_db.c for convenience: */

/* Extract 16-bit big-endian unsigned */
#define GET_U16_BE(data, offset) \
  (((uint16_t)(data)[(offset)] << 8) | (data)[(offset) + 1])

/* Extract 16-bit big-endian signed */
#define GET_S16_BE(data, offset) \
  ((int16_t)GET_U16_BE(data, offset))

/* Extract 32-bit big-endian unsigned */
#define GET_U32_BE(data, offset) \
  (((uint32_t)(data)[(offset)] << 24) | \
   ((uint32_t)(data)[(offset) + 1] << 16) | \
   ((uint32_t)(data)[(offset) + 2] << 8) | \
   (data)[(offset) + 3])
